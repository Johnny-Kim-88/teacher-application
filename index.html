document.getElementById('applyForm').addEventListener('submit', async function(event) { event.preventDefault(); // 1. 필수 입력값 검증 const by = document.getElementById('birth_year').value, bm = document.getElementById('birth_month').value, bd = document.getElementById('birth_day').value; if (!by || !bm || !bd) { alert('생년월일을 올바르게 선택해 주세요.'); return; } const p2 = document.getElementById('phone_mid').value.trim(), p3 = document.getElementById('phone_end').value.trim(); if (!p2 || !p3) { alert('휴대폰 번호를 올바르게 입력해 주세요.'); return; } const btn = document.getElementById('submitBtn'); btn.innerText = '응시원서 전송 중... 잠시만 기다려주세요.'; btn.disabled = true; try { // 2. 화면에 보이지 않게 PDF 데이터 생성 (파일 다운로드 .save() 코드는 실행하지 않음) populatePaperTemplate(); const { jsPDF } = window.jspdf; const pdf = new jsPDF('p', 'mm', 'a4'); const pages = document.querySelectorAll('#pdf-render-area .pdf-page'); for (let i = 0; i < pages.length; i++) { if (i > 0) pdf.addPage(); const canvas = await html2canvas(pages[i], { scale: 2, useCORS: true }); const imgData = canvas.toDataURL('image/jpeg', 1.0); pdf.addImage(imgData, 'JPEG', 0, 0, 210, 297); } // PDF를 Base64 문자열로 변환 (컴퓨터에 저장하지 않고 데이터로만 다룸) const pdfBase64 = pdf.output('datauristring'); const userName = document.getElementById('user_name').value; const userEmail = document.getElementById('user_email').value; const applySubject = document.getElementById('apply_subject').value; // 3. 백엔드 서버(또는 서버리스 함수)로 PDF 데이터 전송 // ※ 1번 방식을 쓰면 '/api/send-email', 2번 방식을 쓰면 'http://localhost:3000/send-email' 입력 const serverUrl = '/api/send-email'; // (2번 로컬 서버라면 'http://localhost:3000/send-email'로 변경) const response = await fetch(serverUrl, { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ user_name: userName, user_email: userEmail, apply_subject: applySubject, pdfBase64: pdfBase64 }) }); const result = await response.json(); if (response.ok && result.success) { alert('응시원서 및 자기소개서가 성공적으로 이메일로 전송되었습니다!'); // 필요시 페이지 초기화 등 추가 가능 } else { throw new Error(result.error || '전송 실패'); } } catch (error) { console.error('전송 오류:', error); alert('이메일 전송 중 오류가 발생했습니다: ' + error.message); } finally { btn.disabled = false; btn.innerText = '응시원서 제출하기'; } });
